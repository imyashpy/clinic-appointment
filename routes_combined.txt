@router.post("", response_model=ClinicOut, status_code=201)
def create_clinic(
    data: ClinicCreate,
    user: User = Depends(require_roles("super_admin")),
    db: Session = Depends(get_db),
):
    exists = (
        db.query(Clinic)
        .filter(Clinic.name == data.name, Clinic.city == data.city)
        .first()
    )
    if exists:
        raise HTTPException(
            status_code=409, detail="A clinic with this name already exists in this city"
        )

    clinic = Clinic(
        name=data.name, address=data.address, city=data.city, phone=data.phone
    )
    db.add(clinic)
    db.flush()
    db.add(
        AuditLog(
            user_id=user.id,
            action="clinic_created",
            entity_type="clinic",
            entity_id=clinic.id,
            details=f"name={clinic.name}; city={clinic.city}",
        )
    )
    db.commit()
    return clinic

def _active_clinic(db: Session, clinic_id: int) -> Clinic:
    clinic = db.get(Clinic, clinic_id)
    if not clinic or not clinic.is_active:
        raise HTTPException(status_code=404, detail="Clinic not found")

def _new_user(db: Session, data, role: str) -> User:
    if db.query(User).filter(User.phone == data.phone).first():
        raise HTTPException(status_code=409, detail=PHONE_TAKEN)
    user = User(
        name=data.name,
        phone=data.phone,
        email=data.email,
        password_hash=hash_password(data.password),
        role=role,  # set here, never taken from the request
    )
    db.add(user)
    db.flush()
    return user

@router.get("/my", response_model=list[AppointmentOut])
def my_appointments(
    when: Literal["upcoming", "past", "all"] = Query("all"),
    user: User = Depends(require_roles("patient")),
    db: Session = Depends(get_db),
):
    patient = get_my_patient(user, db)
    expire_stale_bookings(db, patient.id)

    now = datetime.now()
    query = db.query(Appointment).filter(Appointment.patient_id == patient.id)

    if when == "upcoming":
        query = query.filter(
            Appointment.status.in_(ACTIVE_STATUSES), Appointment.end_at > now
        ).order_by(Appointment.start_at)
    elif when == "past":
        query = query.filter(
            or_(
                ~Appointment.status.in_(ACTIVE_STATUSES),
                Appointment.end_at <= now,
            )
        ).order_by(Appointment.start_at.desc())
    else:
        query = query.order_by(Appointment.start_at.desc())

    return [to_appointment_out(db, a) for a in query.all()]

@router.post("/otp/verify", response_model=LoginResponse)
def otp_verify(data: OtpVerifyRequest, db: Session = Depends(get_db)):
    phone = normalize_phone(data.phone)
    user = db.query(User).filter(User.phone == phone).first()

    if user is None and not data.name:
        # New number: check the code without using it up, then ask for the name
        verify_otp(db, phone, data.code, consume=False)
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Name is required to create your account",
        )

    verify_otp(db, phone, data.code, consume=True)

    is_new = False
    if user is None:
        user = User(
            name=data.name,
            phone=phone,
            email=data.email,
            password_hash=None,  # patients have no password
            role="patient",  # always patient, never taken from the request
        )
        db.add(user)
        try:
            db.flush()
            db.add(
                Patient(
                    user_id=user.id,
                    full_name=data.name,
                    phone=phone,
                    date_of_birth=data.date_of_birth,
                    gender=data.gender,
                )
            )
            db.commit()
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="This mobile number is already registered",
            )
        is_new = True
    elif user.role != "patient" or not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="This account cannot use OTP login",
        )

    token = create_access_token(user.id, user.role)
    return LoginResponse(
        access_token=token,
        user_id=user.id,
        name=user.name,
        role=user.role,
        is_new_user=is_new,
    )


@router.post("/login", response_model=LoginResponse)
def staff_login(data: LoginRequest, db: Session = Depends(get_db)):
    user = db.query(User).filter(User.phone == data.phone).first()

    # Patients have no password and use OTP. Same message for every failure.
    if (
        not user
        or user.role == "patient"
        or not user.password_hash
        or not verify_password(data.password, user.password_hash)
    ):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid phone or password",
        )

    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="This account is disabled",
        )

    token = create_access_token(user.id, user.role)
    return LoginResponse(
        access_token=token,
        user_id=user.id,
        name=user.name,
        role=user.role,
    )

@router.get("/{doctor_id}/slots", response_model=list[SlotOut])
def doctor_slots(
    doctor_id: int,
    day: date = Query(...),
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user),
):
    if not db.get(Doctor, doctor_id):
        raise HTTPException(status_code=404, detail="Doctor not found")

    day_start = datetime.combine(day, time.min)
    day_end = day_start + timedelta(days=1)
    now = datetime.now()

    slots = (
        db.query(Slot)
        .filter(
            Slot.doctor_id == doctor_id,
            Slot.start_at >= day_start,
            Slot.start_at < day_end,
            Slot.start_at > now,  # past slots are never shown
            Slot.status != "blocked",
        )
        .order_by(Slot.start_at)
        .all()
    )

    result = []
    for s in slots:
        slot_status = s.status
        # A lock that has expired counts as free again
        if slot_status == "locked" and (s.locked_until is None or s.locked_until <= now):
            slot_status = "available"
        result.append(
            SlotOut(id=s.id, start_at=s.start_at, end_at=s.end_at, status=slot_status)
        )
    return result

def _age(dob: date | None) -> int | None:
    if not dob:
        return None
    today = date.today()
    return today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))


@router.get("", response_model=list[DoctorAppointmentOut])
def my_day(
    day: date | None = Query(None, description="Defaults to today"),
    doctor: Doctor = Depends(get_my_doctor),
    db: Session = Depends(get_db),
):
    expire_stale_bookings(db)

    day = day or date.today()
    day_start = datetime.combine(day, time.min)
    day_end = day_start + timedelta(days=1)

    rows = (
        db.query(Appointment, Patient)
        .join(Patient, Patient.id == Appointment.patient_id)
        .filter(
            Appointment.doctor_id == doctor.id,
            Appointment.start_at >= day_start,
            Appointment.start_at < day_end,
            # Bookings that were never paid never reached the doctor
            Appointment.payment_status != "pending",
        )
        .order_by(Appointment.start_at)

    )

from datetime import date, datetime, time, timedelta
from typing import Literal

from fastapi import APIRouter, Depends, Query

from sqlalchemy.orm import Session

from app.core.database import get_db
from app.models import Appointment, Doctor, Payment, Refund

router = APIRouter(prefix="/doctors/me/earnings", tags=["Doctor Earnings"])

OPEN_STATUSES = ["confirmed", "arrived", "waiting", "in_consultation", "completed"]


def _range(period: str, anchor: date) -> tuple[date, date]:
    if period == "day":
        return anchor, anchor
    if period == "week":
        start = anchor - timedelta(days=anchor.weekday())  # Monday
        return start, start + timedelta(days=6)
    start = anchor.replace(day=1)
    next_month = (start + timedelta(days=32)).replace(day=1)
    return start, next_month - timedelta(days=1)


@router.get("", response_model=EarningsOut)
def earnings(
    period: Literal["day", "week", "month"] = Query("day"),
    day: date | None = Query(None, description="Any day inside the period, defaults to today"),
    doctor: Doctor = Depends(get_my_doctor),
    db: Session = Depends(get_db),
):
    start_day, end_day = _range(period, day or date.today())
    start = datetime.combine(start_day, time.min)
    end = datetime.combine(end_day, time.min) + timedelta(days=1)

    # Money received in the period. A payment later refunded still counts as received.
    rows = (
        db.query(
            Payment.payment_type,
            Payment.method,
            func.coalesce(func.sum(Payment.amount), 0),
        )
        .join(Appointment, Appointment.id == Payment.appointment_id)
        .filter(
            Appointment.doctor_id == doctor.id,
            Payment.paid_at >= start,
            Payment.paid_at < end,
            Payment.status.in_(["success", "refunded"]),
            Payment.payment_type.in_(["token", "remaining"]),
        )
        .group_by(Payment.payment_type, Payment.method)
        .all()
    )
    token = remaining = online = cash = upi = 0
    for payment_type, method, total in rows:
        total = int(total)
        if payment_type == "token":
            token += total
        else:
            remaining += total
        if method == "online":
            online += total
        elif method == "cash":
            cash += total
        elif method == "upi":
            upi += total

    refunds = int(
        db.query(func.coalesce(func.sum(Refund.amount), 0))
        .join(Appointment, Appointment.id == Refund.appointment_id)
        .filter(
            Appointment.doctor_id == doctor.id,
            Refund.status == "processed",
            Refund.processed_at >= start,
            Refund.processed_at < end,
        )
        .scalar()
    )

    appointments = (
        db.query(Appointment)
        .filter(
            Appointment.doctor_id == doctor.id,
            Appointment.start_at >= start,
            Appointment.start_at < end,
        )
        .all()
    )

    completed = [a for a in appointments if a.status == "completed"]
    no_shows = [a for a in appointments if a.status == "no_show"]
    cancelled = [a for a in appointments if a.status == "cancelled"]

    outstanding = sum(
        a.consultation_fee - a.token_amount
        for a in appointments
        if a.payment_status == "token_paid" and a.status in OPEN_STATUSES
    )

    total_collected = token + remaining
    return EarningsOut(
        period=period,
        start=start_day,
        end=end_day,
        token_collected=token,
        remaining_collected=remaining,
        online_collected=online,
        cash_collected=cash,
        upi_collected=upi,
        total_collected=total_collected,
        refunds=refunds,
        net_collected=total_collected - refunds,
        consultation_revenue=sum(a.consultation_fee for a in completed),
        forfeited_tokens=sum(
            a.token_amount for a in no_shows if a.payment_status == "token_paid"
        ),
        outstanding=outstanding,
        appointments_total=len(appointments),
        completed=len(completed),
        no_show=len(no_shows),
        cancelled=len(cancelled),
    )

@router.get("/dashboard", response_model=ReceptionDashboard)
def dashboard(
    day: date | None = Query(None, description="Defaults to today"),
    user: User = Depends(require_roles("receptionist")),
    db: Session = Depends(get_db),
):
    receptionist = _my_receptionist(db, user)
    expire_stale_bookings(db)
    day, day_start, day_end = _day_bounds(day)

    query = db.query(Appointment).filter(
        Appointment.clinic_id == receptionist.clinic_id,
        Appointment.start_at >= day_start,
        Appointment.start_at < day_end,
        Appointment.payment_status != "pending",
    )
    if receptionist.doctor_id:
        query = query.filter(Appointment.doctor_id == receptionist.doctor_id)
    appointments = query.all()

    counts = Counter(a.status for a in appointments)

    money = {"token": 0, "remaining": 0}
    ids = [a.id for a in appointments]
    if ids:
        rows = (
            db.query(Payment.payment_type, func.coalesce(func.sum(Payment.amount), 0))
            .filter(Payment.appointment_id.in_(ids), Payment.status == "success")
            .group_by(Payment.payment_type)
            .all()
        )
        for payment_type, total in rows:
            if payment_type in money:
                money[payment_type] = int(total)

    remaining_pending = sum(
        a.consultation_fee - a.token_amount
        for a in appointments
        if a.payment_status == "token_paid"
        and a.status in ("confirmed", "arrived", "waiting", "in_consultation", "completed")
    )

    return ReceptionDashboard(
        day=day,
        total_appointments=len(appointments),
        arrived=(
            counts["arrived"]
            + counts["waiting"]
            + counts["in_consultation"]
            + counts["completed"]
        ),
        waiting=counts["arrived"] + counts["waiting"],
        in_consultation=counts["in_consultation"],
        completed=counts["completed"],
        not_arrived=counts["confirmed"],
        no_show=counts["no_show"],
        cancelled=counts["cancelled"],
        token_collected=money["token"],
        remaining_collected=money["remaining"],
        total_received=money["token"] + money["remaining"],
        remaining_pending=remaining_pending,
    )

from datetime import date

from fastapi import APIRouter, Depends, Query
from pydantic import BaseModel
from sqlalchemy import or_
from sqlalchemy.orm import Session

from app.api.deps import require_roles
from app.core.database import get_db
from app.models import Patient, User

router = APIRouter(prefix="/reception/patients", tags=["Reception"])


class PatientSearchOut(BaseModel):
    id: int
    full_name: str
    phone: str
    age: int | None
    gender: str | None
    has_account: bool  # false for walk-in patients with no login


def _age(dob: date | None) -> int | None:
    if not dob:
        return None
    today = date.today()
    return today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))


@router.get("", response_model=list[PatientSearchOut])
def search_patients(
    q: str = Query(..., min_length=3, max_length=50, description="Name or mobile number"),
    user: User = Depends(require_roles("receptionist")),
    db: Session = Depends(get_db),
):
    q = q.strip()
    # Treat % and _ as normal characters, not as wildcards
    safe = q.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")
    pattern = f"%{safe}%"

    patients = (
        db.query(Patient)
        .filter(
            or_(
                Patient.full_name.ilike(pattern, escape="\\"),
                Patient.phone.like(pattern, escape="\\"),
            )
        )
        .order_by(Patient.full_name)
        .limit(20)
        .all()
    )

    return [
        PatientSearchOut(
            id=p.id,
            full_name=p.full_name,
            phone=p.phone,
            age=_age(p.date_of_birth),
            gender=p.gender,
            has_account=p.user_id is not None,
        )
        for p in patients
    ]

from fastapi import APIRouter, Depends, HTTPException, status

from app.models import Doctor, DoctorSchedule, User
from app.schemas.schedule import ScheduleDay, ScheduleUpdate

router = APIRouter(prefix="/doctors/me/schedule", tags=["Doctor Schedule"])


def get_my_doctor(
    user: User = Depends(require_roles("doctor")),
    db: Session = Depends(get_db),
) -> Doctor:
    doctor = db.query(Doctor).filter(Doctor.user_id == user.id).first()
    if not doctor:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Doctor profile not found",
        )
    return doctor


@router.get("", response_model=list[ScheduleDay])
def get_schedule(
    doctor: Doctor = Depends(get_my_doctor),
    db: Session = Depends(get_db),
):
    return (
        db.query(DoctorSchedule)
        .filter(DoctorSchedule.doctor_id == doctor.id)
        .order_by(DoctorSchedule.day_of_week)
        .all()
    )


@router.put("", response_model=list[ScheduleDay])
def set_schedule(
    data: ScheduleUpdate,
    doctor: Doctor = Depends(get_my_doctor),
    db: Session = Depends(get_db),
):
    # Replace the whole weekly schedule with what the doctor sent
    db.query(DoctorSchedule).filter(DoctorSchedule.doctor_id == doctor.id).delete()
    for day in data.days:
        db.add(DoctorSchedule(doctor_id=doctor.id, **day.model_dump()))
    db.commit()

    return (
        db.query(DoctorSchedule)
        .filter(DoctorSchedule.doctor_id == doctor.id)
        .order_by(DoctorSchedule.day_of_week)
        .all()
    )
